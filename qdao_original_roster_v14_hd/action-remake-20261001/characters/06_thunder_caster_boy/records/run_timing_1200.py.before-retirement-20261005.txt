import json
from pathlib import Path
ROOT=Path(__file__).resolve().parents[1]
def update():
 for d in ['E','W','NE']:
  p=ROOT/'review'/f'run_{d}_preview.html'
  if p.exists():
   s=p.read_text(encoding='utf-8-sig').replace('480ms为旧基线；新节奏640/720/800对比见总预览。','当前统一1200ms/圈，每帧75ms；另有慢放、暂停与逐帧。').replace('480 / 640 / 720 / 800 对比预览','1200ms正式节奏预览').replace('旧基线30ms','正常75ms · 1200ms/圈').replace('s.ms=30;','s.ms=75;')
   p.write_text(s,encoding='utf-8')
  p=ROOT/'review'/f'run_{d}_grounding_phase_20261003.json'
  if p.exists():
   r=json.loads(p.read_text(encoding='utf-8-sig'))
   for key in ['timingSuggestion','suggestedTiming']:
    if key in r:
     r['deprecated_'+key]=r.pop(key)
     r['deprecated_'+key]['status']='已过期，不应用：用户最新指定1200ms/圈，16×75ms。历史候选保留仅供追溯。'
   r['currentTiming']={'frameMs':75,'frameCount':16,'loopMs':1200,'source':'用户最新明确参数，经root转述2026-10-03','visualAcceptance':False,'clientAcceptance':False}
   for f in r.get('frames',[]):
    if 'suggestedHoldMsAfterAcceptance' in f:f['deprecatedSuggestedHoldMs']=f.pop('suggestedHoldMsAfterAcceptance')
    f['currentFrameMs']=75
   p.write_text(json.dumps(r,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
  p=ROOT/'review'/f'run_{d}_preview.generation.json'
  if p.exists():
   r=json.loads(p.read_text(encoding='utf-8-sig'));r['currentTiming']={'requestedFrameMs':75,'loopMsFor16Frames':1200,'gifEncoding':'GIF uses alternating70/80ms due10ms timebase; HTML75ms. This does not imply dynamics accepted.'}
   p.write_text(json.dumps(r,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
if __name__=='__main__':update()
